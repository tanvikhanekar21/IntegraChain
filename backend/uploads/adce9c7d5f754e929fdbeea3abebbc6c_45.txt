# Feature selection pipeline
import pandas as pd
import numpy as np
from sklearn.feature_selection import VarianceThreshold, SelectKBest, f_classif
from sklearn.preprocessing import LabelEncoder

# Load dataset
df = pd.read_csv("45_dataset.csv")
print(f"loaded dataset with shape:{df.shape}")
print(df.head(10))

# Drop identifier columns
df = df.drop(columns=["iso_code", "location", "date"])

# Split into X and Y
X = df.drop(columns=["continent"])
Y = df["continent"]
print(X)
print(Y)

# Encode categorical features
cat_cols = X.select_dtypes(include=["object", "category", "string"]).columns
encoders = {}
for col in cat_cols:
    le = LabelEncoder()
    X[col] = le.fit_transform(X[col].fillna("Unknown").astype(str))
    encoders[col] = le
    print(f"encoded categorical columns:{list(cat_cols)}")
print(X.head(5))

# Fill null values
print("\nNull values before filling:")
print(X.isnull().sum())
for col in X.columns:
    X[col] = X[col].fillna(X[col].median())
print("\nNull values after filling:")
print(X.isnull().sum())

# Remove duplicates
print(f"\nDuplicate rows before removal:{X.duplicated().sum()}")
duplicate_rows = X.duplicated()
X = X[~duplicate_rows].reset_index(drop=True)
Y = Y[~duplicate_rows].reset_index(drop=True)
print(f"Duplicate rows after removal:{X.duplicated().sum()}")

# Encode target variable
Y = Y.fillna(Y.mode()[0])
le_target = LabelEncoder()
Y = le_target.fit_transform(Y.astype(str))
print("\nTarget variable encoded successfully")

# Remove low-variance features
print("\ncolumns before removal::")
print(X.columns.tolist())
selector = VarianceThreshold(threshold=0.0)
selector.fit(X)
kept_cols = X.columns[selector.get_support()]
print(f"variance threshold:kept {len(kept_cols)}/{X.shape[1]} features")
print("columns after removal::")
print(kept_cols.tolist())
cols_to_drop = [col for col in X.columns if col not in kept_cols]
X = X.drop(columns=cols_to_drop)
print(X.columns.tolist())

# Remove highly correlated features
corr_matrix = X.corr().abs()
upper = corr_matrix.where(
    np.triu(np.ones(corr_matrix.shape), k=1).astype(bool)
)
to_drop = [
    col for col in upper.columns
    if any(upper[col] > 0.90)
]
print(f"correlation filter (>0.90): dropping {to_drop if to_drop else 'none'}")
X = X.drop(columns=to_drop)
print("after removing highly correlated features::")
print(X.columns.tolist())

# ANOVA feature selection
k = min(5, X.shape[1])
selector = SelectKBest(score_func=f_classif, k=k)
selector.fit(X, Y)
scores = pd.Series(
    selector.scores_,
    index=X.columns
).sort_values(ascending=False)
print(f"\nSelectKBest (ANOVA F-test)scores,ranked::\n{scores}")
kept_cols = scores.head(k).index
to_drop = [col for col in X.columns if col not in kept_cols]
X = X.drop(columns=to_drop)
print("after removing columns using ANOVA::")
print(X.columns.tolist())

print("\nFinal selected features:")
print(X.head(10))
print("\nFinal feature count:", X.shape[1])
